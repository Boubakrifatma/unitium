# Prédiction simple
curl -X POST http://localhost:5000/predict \
  -H "Content-Type: application/json" \
  -d '{
    "created_date": "2026-05-01",
    "due_date": "2026-05-15",
    "priority": "High",
    "category": "Development",
    "estimated_hours": 40
  }'
